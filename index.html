analysis_co2_historical_rankings.sql 
WITH annual_aggregates AS (
    -- Filtrado y extracción de métricas principales normalizadas
    SELECT 
        country,
        year,
        co2 AS territorial_co2_mt,
        co2_per_capita,
        cumulative_co2,
        -- Tasa de crecimiento interanual con función de ventana LAG()
        LAG(co2, 1) OVER (PARTITION BY country ORDER BY year) AS prev_co2
    FROM raw_owid_emissions
    WHERE iso_code IS NOT NULL -- Excluye agregados continentales brutos
),
debt_metrics AS (
    SELECT 
        country,
        year,
        territorial_co2_mt,
        co2_per_capita,
        cumulative_co2,
        ROUND(
            ((territorial_co2_mt - prev_co2) / NULLIF(prev_co2, 0)) * 100, 
            2
        ) AS yoy_emission_change_pct,
        -- Porcentaje de cuota histórica frente al acumulado mundial en el año de corte
        DENSE_RANK() OVER (PARTITION BY year ORDER BY cumulative_co2 DESC) AS historical_rank
    FROM annual_aggregates
)
SELECT 
    country,
    year,
    territorial_co2_mt,
    co2_per_capita,
    cumulative_co2,
    yoy_emission_change_pct,
    historical_rank
FROM debt_metrics
WHERE year = 2023 AND historical_rank <= 10
ORDER BY historical_rank ASC;
